# A one-screen test app, separate from the real one.
#
# The reader that fills the Win Rate Rank table has never seen Chamet's own
# rank screen - it is written against what that screen ought to look like in
# the view tree, which is a guess until somebody points it at the real thing.
# Shipping a guess into the app that prices every bet is the wrong order to
# do this in.
#
# So this builds a tiny separate app that does the reading and prints the
# result, and nothing else: no WebView, no database, no network, no signing
# key, and a different package id so it installs alongside rather than over
# the real app. If the table comes back right, the same parser goes into the
# app unchanged - it is copied from the app's own source here, not rewritten.
# If it comes back wrong, the Copy button hands over the actual view tree and
# the parser gets fixed against that instead of against a guess.
#
# Run it by hand from the Actions tab - it deliberately does not fire on push.

          package com.racexai.rankdemo;

          import java.util.ArrayList;
          import java.util.LinkedHashSet;
          import java.util.List;

          /**
           * Turns one dumped frame into the handful of facts the bot actually needs, and
           * works out when a race starts and ends.
           *
           * This exists because shipping 400KB of view tree back and forth does not work -
           * it gets truncated in transit and the interesting moment is never in the part
           * that survives. The phone can do the reading itself and report a dozen lines.
           *
           * Deliberately free of Android imports so the parsing can be run and checked on
           * a desktop against real dumps before it ever reaches a device.
           */
          public class RaceReader {

              /**
               * Longest first, on purpose. "Sports Car" has to be tested before "Car" or
               * the shorter name matches inside the longer one and every Sports Car win is
               * recorded as a Car win.
               */
              private static final String[] VEHICLES = {
                  "Monster Truck", "Sports Car", "Stock Car", "Supercar",
                  "Motorcycle", "ATV", "ORV", "SUV", "Car"
              };

              /**
               * The six surfaces. Needed here so a track sign can be told apart from the
               * "???" placeholder on the two tracks the game keeps back - matching a known
               * name is the only reliable test, since the placeholder text varies by
               * locale and by build.
               */
              private static final String[] ROADS = {
                  "Expressway", "Potholes", "Highway", "Desert", "Bumpy", "Dirt"
              };

              public static class Frame {
                  public final List<String> cars = new ArrayList<>();   // left to right
                  public final List<String> odds = new ArrayList<>();
                  public final List<String> bets = new ArrayList<>();   // crowd stake per vehicle
                  public String road;              // the one revealed track
                  public int slot = -1;            // 1..3, which track is revealed
                  public String countdown;
                  public String winner;            // from the result banner: "SUV won, Next 8 s"
                  public boolean betOpen;          // betting panel has height => betting is open
                  public final LinkedHashSet<String> pairs = new LinkedHashSet<>();  // "id=text"

                  /**
                   * Every track sign on screen, left to right, exactly as the view tree
                   * gives them - "???" placeholders included, and each one's content
                   * description appended after a | if it has one.
                   *
                   * Knowing all three surfaces before the bet is worth about seventeen
                   * points of accuracy, so the one question worth keeping open is whether
                   * the game's own client is holding the hidden two and merely declining
                   * to draw them. A screen reader would be told if it were. This records
                   * what is actually there, every race, so the question gets settled by
                   * evidence rather than by assuming the answer.
                   */
                  public final List<String> signs = new ArrayList<>();

                  /**
                   * The signs that name a real surface, left to right, with an empty
                   * string wherever one still reads "???". Once the race starts the game
                   * fills all three in, and that is the only moment the hidden two are
                   * ever visible.
                   */
                  public final List<String> namedRoads = new ArrayList<>();
              }

              /**
               * Was the last board read from the view ids we know, or from the words on
               * screen? Shown in the diagnostic, so "it works on ChillChat" is a fact
               * about which route got there rather than a guess.
               */
              public static volatile boolean byContent = false;

              public static Frame parse(String dump) {
                  byContent = false;
                  Frame f = new Frame();
                  List<int[]> tracks = new ArrayList<>();      // {left, hasSign}
                  List<String[]> carsAt = new ArrayList<>();   // {left, name}
                  List<String[]> oddsAt = new ArrayList<>();
                  List<String[]> betsAt = new ArrayList<>();
                  List<String[]> signsAt = new ArrayList<>();  // {left, text|desc}

                  for (String line : dump.split("\n")) {
                      int p = line.indexOf("id=");
                      if (p < 0) continue;
                      int q = p + 3;
                      while (q < line.length() && line.charAt(q) != ' ' && line.charAt(q) != '\n') q++;
                      String vid = line.substring(p + 3, q);
                      int slash = vid.lastIndexOf(":id/");
                      if (slash >= 0) vid = vid.substring(slash + 4);

                      String text = between(line, "TEXT=\"", "\"");
                      String desc = between(line, "DESC=\"", "\"");
                      int[] b = bounds(line);

                      if (text != null && text.length() > 0) {
                          f.pairs.add(vid + "=" + text);
                          if (f.winner == null) f.winner = winnerIn(text);
                      }

                      if (vid.equals("gameBetView") && b != null) f.betOpen = (b[3] - b[1]) > 50;
                      else if (vid.equals("tv_car_name") && text != null)
                          carsAt.add(new String[]{ String.valueOf(b == null ? 0 : b[0]), text });
                      else if (vid.equals("tv_car_odds") && text != null)
                          oddsAt.add(new String[]{ String.valueOf(b == null ? 0 : b[0]), text });
                      else if (vid.equals("tv_totalBet") && text != null)
                          betsAt.add(new String[]{ String.valueOf(b == null ? 0 : b[0]), text });
                      else if (vid.equals("tv_track_sign") && (text != null || desc != null)) {
                          // Record every sign, then pick the revealed road from whichever
                          // one names a surface we know. Taking the last sign to carry any
                          // text at all - which is what this did - hands back "???" the
                          // moment the placeholders turn out to share this id, and the
                          // whole board is then filled against the wrong road.
                          signsAt.add(new String[]{ String.valueOf(b == null ? 0 : b[0]),
                                                    (text == null ? "" : text)
                                                    + (desc == null ? "" : "|" + desc) });
                          String known = roadIn(text);
                          if (known == null) known = roadIn(desc);
                          if (known != null) f.road = known;
                      }
                      else if (vid.equals("tv_race_surplus") && text != null) f.countdown = text;
                      else if (vid.equals("cl_track_content") && b != null) tracks.add(new int[]{ b[0], 0 });
                      else if (vid.equals("ll_track_sign") && b != null) {
                          // the revealed track is whichever track box this sign sits inside
                          for (int[] t : tracks) if (b[0] >= t[0] - 20 && b[0] <= t[0] + 400) t[1] = 1;
                      }
                  }

                  // ── WHEN THE IDS ARE NOT THE ONES WE KNOW ──
                  //
                  // Everything above is keyed off exact view ids - tv_car_name,
                  // tv_car_odds, tv_track_sign. On the build those came from that is
                  // precise and fast. On a build that names its views anything else it
                  // finds nothing at all, silently, and the bot sits there doing nothing
                  // with no way to say why. That is the same mistake as reading one
                  // hardcoded package name, one layer further in.
                  //
                  // A view id is a label. The CONTENT is the thing. "Monster Truck" is a
                  // vehicle name whatever the box holding it is called, and there are only
                  // nine of them. So when the ids find no board, the names are looked for
                  // directly - and because a chat message could easily contain the word
                  // "SUV", the text has to be EXACTLY a vehicle name and there have to be
                  // three of them sitting in a row.
                  if (carsAt.size() < 3) {
                      byContent = readByContent(dump, carsAt, oddsAt, signsAt, f);
                      // WHETHER BETTING IS OPEN, on a build that has no gameBetView.
                      //
                      // betOpen is measured from the height of one particular view, and a
                      // build that does not have that view leaves it false forever - so
                      // the watcher never fills a board it read perfectly well. The view
                      // was only ever a proxy for the real question anyway: the board is
                      // on screen and the race has not finished. When there is no view to
                      // ask, ask that instead.
                      if (byContent && dump.indexOf("gameBetView") < 0)
                          f.betOpen = (carsAt.size() == 3 && f.winner == null);
                  }

                  sortByLeft(carsAt);
                  sortByLeft(oddsAt);
                  sortByLeft(betsAt);
                  sortByLeft(signsAt);
                  for (String[] c : carsAt) f.cars.add(c[1]);
                  for (String[] o : oddsAt) f.odds.add(o[1]);
                  for (String[] o : betsAt) f.bets.add(o[1]);
                  for (String[] s : signsAt) f.signs.add(s[1]);
                  // the same signs, resolved to surface names where they name one
                  for (String[] s : signsAt) {
                      String r = roadIn(s[1]);
                      f.namedRoads.add(r == null ? "" : r);
                  }
                  // a board with a "???" on it has not revealed anything
                  for (String r : f.namedRoads) if (r.isEmpty()) { f.namedRoads.clear(); break; }

                  // tracks in screen order; the one carrying a sign is the revealed slot
                  for (int i = 0; i < tracks.size(); i++)
                      for (int j = i + 1; j < tracks.size(); j++)
                          if (tracks.get(j)[0] < tracks.get(i)[0]) {
                              int[] tmp = tracks.get(i); tracks.set(i, tracks.get(j)); tracks.set(j, tmp);
                          }
                  for (int i = 0; i < tracks.size(); i++) if (tracks.get(i)[1] == 1) f.slot = i + 1;

                  // THE SIGNS THEMSELVES SAY WHICH SLOT IT IS.
                  //
                  // Above, the slot comes from which track BOX contains a sign - which
                  // needs cl_track_content and ll_track_sign to be named exactly that. But
                  // the answer is already sitting in the signs: two of them read "???" and
                  // one names a road, and the one that names a road is the revealed slot.
                  // No ids required, and it agrees with the box method wherever both work.
                  if (f.slot < 1 && signsAt.size() >= 2) {
                      int named = -1, count = 0;
                      for (int i = 0; i < signsAt.size(); i++) {
                          if (roadIn(signsAt.get(i)[1]) != null) { named = i; count++; }
                      }
                      if (count == 1) f.slot = named + 1;
                  }

                  return f;
              }

              /**
               * Find the board by what it says rather than by what its views are called.
               *
               * Returns true if a board was actually found this way. Deliberately strict:
               * a wrong board is far worse than no board, because a wrong one gets logged
               * and teaches the models something untrue, while no board just means the bot
               * waits. So the vehicle name has to match a known one exactly, three of them
               * have to sit in the same horizontal band, and odds are only taken when each
               * car has exactly one plausible number lined up beneath it.
               */
              private static boolean readByContent(String dump, List<String[]> carsAt,
                                                   List<String[]> oddsAt, List<String[]> signsAt,
                                                   Frame f) {
                  List<int[]> carXY = new ArrayList<>();       // {x, y}
                  List<String> carName = new ArrayList<>();
                  List<int[]> numXY = new ArrayList<>();       // {x, y}
                  List<String> numText = new ArrayList<>();
                  List<int[]> signXY = new ArrayList<>();
                  List<String> signText = new ArrayList<>();

                  for (String line : dump.split("\n")) {
                      String text = between(line, "TEXT=\"", "\"");
                      String desc = between(line, "DESC=\"", "\"");
                      int[] b = bounds(line);
                      if (b == null) continue;
                      int cx = (b[0] + b[2]) / 2, cy = (b[1] + b[3]) / 2;

                      String v = exactVehicle(text);
                      if (v == null) v = exactVehicle(desc);
                      if (v != null) { carXY.add(new int[]{ cx, cy }); carName.add(v); continue; }

                      String r = exactRoad(text);
                      if (r == null) r = exactRoad(desc);
                      if (r != null || isPlaceholder(text)) {
                          signXY.add(new int[]{ cx, cy });
                          signText.add(r != null ? r : (text == null ? "" : text));
                          continue;
                      }

                      if (looksLikeOdds(text)) { numXY.add(new int[]{ cx, cy }); numText.add(text); }
                  }

                  if (carName.size() < 3) return false;

                  // THREE IN A ROW. A board draws its three vehicles side by side, so the
                  // right three share a horizontal band. Anything scattered down the
                  // screen is a chat message that happens to contain a vehicle name.
                  int bestStart = -1, bestCount = 0, bestY = 0;
                  for (int i = 0; i < carXY.size(); i++) {
                      int n = 0, sum = 0;
                      for (int j = 0; j < carXY.size(); j++)
                          if (Math.abs(carXY.get(j)[1] - carXY.get(i)[1]) <= 120) { n++; sum += carXY.get(j)[1]; }
                      if (n > bestCount) { bestCount = n; bestStart = i; bestY = sum / n; }
                  }
                  if (bestCount < 3) return false;

                  List<Integer> pick = new ArrayList<>();
                  for (int j = 0; j < carXY.size(); j++)
                      if (Math.abs(carXY.get(j)[1] - bestY) <= 120) pick.add(j);
                  // more than three means something else on screen is named like a vehicle
                  if (pick.size() != 3) return false;

                  for (int j : pick)
                      carsAt.add(new String[]{ String.valueOf(carXY.get(j)[0]), carName.get(j) });

                  // ODDS, only when each car has exactly one number under it.
                  //
                  // Half a set of odds is worse than none: the app would pair them with
                  // the wrong vehicles and a model would be fitted to nonsense. All three
                  // or nothing.
                  List<String[]> got = new ArrayList<>();
                  for (int j : pick) {
                      int cx = carXY.get(j)[0], hits = 0;
                      String best = null;
                      for (int k = 0; k < numXY.size(); k++) {
                          int dy = numXY.get(k)[1] - bestY;
                          if (dy < -40 || dy > 320) continue;               // beneath the name
                          if (Math.abs(numXY.get(k)[0] - cx) > 90) continue; // in its column
                          hits++; best = numText.get(k);
                      }
                      if (hits == 1) got.add(new String[]{ String.valueOf(cx), best });
                  }
                  if (got.size() == 3) oddsAt.addAll(got);

                  // the track signs, if the ids did not already find them
                  if (signsAt.isEmpty() && signText.size() >= 2) {
                      for (int i = 0; i < signText.size(); i++) {
                          signsAt.add(new String[]{ String.valueOf(signXY.get(i)[0]), signText.get(i) });
                          String known = roadIn(signText.get(i));
                          if (known != null && f.road == null) f.road = known;
                      }
                  }
                  return true;
              }

              /**
               * Is there a betting board in this dump, judged by content alone?
               *
               * Used by the watcher to decide whether a screen is worth reading at all,
               * on builds whose view ids we do not recognise. Runs the real parser rather
               * than a cheaper approximation, because "the parser can read it" is the only
               * definition of "there is a board here" that is actually useful - a test
               * that says yes to something the parser then cannot read would just move the
               * silence one step later.
               */
              public static boolean looksLikeBoard(String dump) {
                  try {
                      Frame f = parse(dump);
                      return f.cars.size() == 3;
                  } catch (Exception e) { return false; }
              }

              /* ══════════════════════════════════════════════════════════════════
                 THE GAME'S OWN WIN RATE RANK SCREEN

                 Nine rows, each a vehicle with its win rate and its payout. Open it
                 in Chamet and the eighteen numbers are right there on screen - so
                 there is no good reason to make anybody copy them into the app by
                 hand, twice, squinting between two halves of a split screen.

                 WHAT THIS DOES NOT DO IS TRUST THE LAYOUT. No view ids are used at
                 all: this build names them one thing, the next build names them
                 another, and a rank reader that silently returns nothing on a
                 renamed screen is worse than none. It finds the nine vehicle names
                 by their words, then takes the numbers that sit on the same ROW as
                 each - which is a fact about the picture, not about the code that
                 drew it.

                 THE ACCEPTANCE TEST IS THE INTERESTING PART. Any screen with words
                 and numbers on it can be mispaired into a plausible-looking table,
                 and a wrong rank table would poison every price in the app while
                 looking perfectly filled in. So the pairing has to prove itself:

                     win rate x payout ~= 0.98,  for every row

                 That is not a coincidence to be checked against - it is how the
                 game prices the thing. It keeps about 2% and hands back the rest.
                 A row where the two numbers came from different places will not
                 land near 0.98, and seven rows in a row landing there by accident
                 is not something that happens. If the margins do not agree, this
                 returns nothing and says why.
                 ══════════════════════════════════════════════════════════════════ */

              public static class RankRow {
                  public final String vehicle;
                  public final double winPct;   // 39.4
                  public final double payout;   // 2.5
                  RankRow(String v, double w, double p) { vehicle = v; winPct = w; payout = p; }
                  public double margin() { return winPct / 100.0 * payout; }
              }

              public static class Rank {
                  public final List<RankRow> rows = new ArrayList<>();
                  /** mean of winRate x payout across the rows - the game's own take */
                  public double margin;
                  /** why nothing was returned, when nothing was */
                  public String why = "no rank screen here";
              }

              private static final int RANK_MIN_ROWS = 7;      // of nine
              private static final double MARGIN_LO = 0.80;    // the game keeps 2%; allow
              private static final double MARGIN_HI = 1.15;    // for rounding and odd builds

              /**
               * Cheap enough to run on every frame. The betting board names three
               * vehicles; only the rank screen names seven or more.
               */
              public static int vehicleLabels(String dump) {
                  LinkedHashSet<String> seen = new LinkedHashSet<>();
                  for (String line : dump.split("\n")) {
                      String v = exactVehicle(between(line, "TEXT=\"", "\""));
                      if (v == null) v = exactVehicle(between(line, "DESC=\"", "\""));
                      if (v != null) seen.add(v);
                  }
                  return seen.size();
              }

              public static boolean looksLikeRank(String dump) {
                  return vehicleLabels(dump) >= RANK_MIN_ROWS && parseRank(dump).rows.size() >= RANK_MIN_ROWS;
              }

              public static Rank parseRank(String dump) {
                  Rank out = new Rank();
                  List<double[]> nums = new ArrayList<>();      // {y, value, used}
                  List<Object[]> names = new ArrayList<>();     // {y, name}

                  for (String line : dump.split("\n")) {
                      int[] b = bounds(line);
                      if (b == null) continue;
                      double y = (b[1] + b[3]) / 2.0;
                      String text = between(line, "TEXT=\"", "\"");
                      if (text == null || text.trim().length() == 0) text = between(line, "DESC=\"", "\"");
                      if (text == null) continue;
                      String v = exactVehicle(text);
                      if (v != null) { names.add(new Object[]{ y, v }); continue; }
                      double n = number(text);
                      if (!Double.isNaN(n)) nums.add(new double[]{ y, n, 0 });
                  }
                  if (names.size() < RANK_MIN_ROWS) { out.why = "only " + names.size() + " vehicles named"; return out; }

                  // How tall is a row? Taken from the gaps between the vehicle names
                  // themselves rather than typed in, so it holds at any screen density.
                  double tol = rowTolerance(names);

                  // Two numbers per vehicle. Which is the win rate and which the payout
                  // is decided by size, not by column order: a payout is a small
                  // multiplier, a win rate is a percentage. Builds that swap the columns
                  // round therefore read the same.
                  List<RankRow> got = new ArrayList<>();
                  LinkedHashSet<String> done = new LinkedHashSet<>();
                  for (Object[] nm : names) {
                      double y = (Double) nm[0];
                      String v = (String) nm[1];
                      if (done.contains(v)) continue;
                      double[] a = nearest(nums, y, tol, null);
                      if (a == null) continue;
                      double[] c = nearest(nums, y, tol, a);
                      if (c == null) { a[2] = 0; continue; }
                      double hi = Math.max(a[1], c[1]), lo = Math.min(a[1], c[1]);
                      if (!(lo >= 1.0 && lo <= 30.0)) { a[2] = 0; c[2] = 0; continue; }   // payout
                      if (!(hi >= 3.0 && hi <= 80.0)) { a[2] = 0; c[2] = 0; continue; }   // win %
                      done.add(v);
                      got.add(new RankRow(v, hi, lo));
                  }
                  if (got.size() < RANK_MIN_ROWS) { out.why = "only " + got.size() + " rows paired up"; return out; }

                  // ── and now the part that decides whether to believe any of it ──
                  double sum = 0;
                  for (RankRow r : got) sum += r.margin();
                  double mean = sum / got.size();
                  double worst = 0;
                  for (RankRow r : got) worst = Math.max(worst, Math.abs(r.margin() - mean));
                  if (mean < MARGIN_LO || mean > MARGIN_HI) {
                      out.why = "the numbers do not price a race (margin " + round2(mean) + ")";
                      return out;
                  }
                  if (worst > 0.16) {
                      out.why = "rows disagree with each other (worst off by " + round2(worst) + ")";
                      return out;
                  }
                  out.rows.addAll(got);
                  out.margin = mean;
                  out.why = "";
                  return out;
              }

              /** Half the typical gap between one vehicle row and the next. */
              private static double rowTolerance(List<Object[]> names) {
                  List<Double> ys = new ArrayList<>();
                  for (Object[] n : names) ys.add((Double) n[0]);
                  for (int i = 0; i < ys.size(); i++)
                      for (int j = i + 1; j < ys.size(); j++)
                          if (ys.get(j) < ys.get(i)) { double t = ys.get(i); ys.set(i, ys.get(j)); ys.set(j, t); }
                  double best = -1;
                  for (int i = 1; i < ys.size(); i++) {
                      double g = ys.get(i) - ys.get(i - 1);
                      if (g > 1 && (best < 0 || g < best)) best = g;
                  }
                  return best < 0 ? 60 : best * 0.45;
              }

              /** The unused number closest to this row, or null. Marks it used. */
              private static double[] nearest(List<double[]> nums, double y, double tol, double[] skip) {
                  double[] best = null;
                  double bd = 0;
                  for (double[] n : nums) {
                      if (n[2] != 0 || n == skip) continue;
                      double d = Math.abs(n[0] - y);
                      if (d > tol) continue;
                      if (best == null || d < bd) { best = n; bd = d; }
                  }
                  if (best != null) best[2] = 1;
                  return best;
              }

               *
               * Bare integers are refused: a rank screen is full of positions, ranks
               * and counts, and letting "3" become a payout is how a mispaired table
               * gets built. Both real columns carry a decimal point or a % sign.
               */
              static double number(String text) {
                  if (text == null) return Double.NaN;
                  String t = text.trim().toLowerCase();
                  boolean pct = t.endsWith("%");
                  if (pct) t = t.substring(0, t.length() - 1).trim();
                  if (t.startsWith("x")) t = t.substring(1).trim();
                  if (t.endsWith("x")) t = t.substring(0, t.length() - 1).trim();
                  t = t.replace(',', '.');
                  if (t.length() == 0 || (!pct && t.indexOf('.') < 0)) return Double.NaN;
                  for (int i = 0; i < t.length(); i++) {
                      char c = t.charAt(i);
                      if (!Character.isDigit(c) && c != '.') return Double.NaN;
                  }
                  try {
                      double d = Double.parseDouble(t);
                      return (d > 0 && d < 1000) ? d : Double.NaN;
                  } catch (Exception e) { return Double.NaN; }
              }

                  sb.append("\nanything naming a road: ").append(roadish.isEmpty() ? "(none)" : join(roadish, " | "));
                  return sb.toString();

                  for (String s : set) { if (sb.length() > 0) sb.append(sep); sb.append(s); }
